// ============================================================ // MODAL FUNCTIONS (Add/Edit) // ============================================================ function openModal() { document.getElementById('itemModal').classList.add('active'); document.getElementById('modalTitle').textContent = 'Add New Item'; document.getElementById('modalSub').textContent = 'Fill in the details below'; document.getElementById('editId').value = ''; document.getElementById('itemForm').reset(); editingId = null; } function closeModal() { document.getElementById('itemModal').classList.remove('active'); } function saveItem(event) { event.preventDefault(); const name = document.getElementById('itemName').value.trim(); const category = document.getElementById('itemCategory').value; const qty = parseInt(document.getElementById('itemQty').value); const minStock = parseInt(document.getElementById('itemMin').value); const status = document.getElementById('itemStatus').value; const editId = document.getElementById('editId').value; if (!name || !category || isNaN(qty) || isNaN(minStock)) { showToast('error', 'Validation Error', 'Please fill in all fields correctly.'); return; } if (editId) { // Edit existing item const index = inventory.findIndex(i => i.id === parseInt(editId)); if (index !== -1) { inventory[index] = { ...inventory[index], name, category, qty, minStock, status }; showToast('success', 'Item Updated', `${name} has been updated.`); } } else { // Add new item const newItem = { id: idCounter++, name, category, qty, minStock, status }; inventory.push(newItem); showToast('success', 'Item Added', `${name} has been added to inventory.`); } // Save to localStorage localStorage.setItem('visiontory-inventory', JSON.stringify(inventory)); closeModal(); renderAll(); } function editItem(id) { const item = inventory.find(i => i.id === id); if (!item) return; document.getElementById('itemModal').classList.add('active'); document.getElementById('modalTitle').textContent = 'Edit Item'; document.getElementById('modalSub').textContent = 'Update the item details below'; document.getElementById('editId').value = item.id; document.getElementById('itemName').value = item.name; document.getElementById('itemCategory').value = item.category; document.getElementById('itemQty').value = item.qty; document.getElementById('itemMin').value = item.minStock; document.getElementById('itemStatus').value = item.status; editingId = item.id; } function deleteItem(id) { const item = inventory.find(i => i.id === id); if (!item) return; if (confirm(`Are you sure you want to delete "${item.name}"?`)) { inventory = inventory.filter(i => i.id !== id); localStorage.setItem('visiontory-inventory', JSON.stringify(inventory)); renderAll(); showToast('info', 'Item Deleted', `${item.name} has been removed.`); } }
